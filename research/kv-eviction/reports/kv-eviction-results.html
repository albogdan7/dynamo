class EvictionPolicy:
    def select_victim(self, blocks: list[dict]) -> int:
        # Evict blocks that were never reused (fewest releases) first; blocks
        # that have cycled through the inactive queue repeatedly are likely

class EvictionPolicy:
    def __init__(self):
        pass

    def select_victim(self, blocks: list[dict]) -> int:
        # Give copies that have re-entered the inactive queue a second
        # chance. Release counts describe physical-copy lifecycle events,
        # so use only a binary distinction, not a frequency estimate.
        # Prefer the oldest probationary copy whenever one is available;
        # otherwise let native recency age the protected copies out.
        return min(
            blocks,
            key=lambda block: (
                block["num_releases"] > 1,
                block["lru_rank"],
            ),
        )["block_id"]

class EvictionPolicy:
    def select_victim(self, blocks: list[dict]) -> int:
        # Release counters describe physical-copy reuse, not token hit counts.
        # Give repeatedly released copies a bounded retention bonus, measured
        # in logical release events. Aging lets old copies lose that bonus.
        newest_release = max(block["release_order"] for block in blocks)
        horizon = max(1, len(blocks))

        def priority(block):
            age = newest_release - block["release_order"]
            reuse = min(3, max(0, block["num_releases"] - 1))
            return (
                age - horizon * reuse,
                -block["lru_rank"],
            )

        return max(blocks, key=priority)["block_id"]

class EvictionPolicy:
    def select_victim(self, blocks: list[dict]) -> int:
        # Restrict to the older half of the inactive queue so recency still matters,
        # then prefer blocks released least often (likely unshared tails), and among
        # those the most recently created (deeper in a prefix chain), then oldest.
        ordered = sorted(blocks, key=lambda b: b["lru_rank"])
        window = ordered[: max(1, (len(ordered) + 1) // 2)]
        best = min(
            window,
            key=lambda b: (b["num_releases"], -b["creation_order"], b["lru_rank"]),
        )
        return best["block_id"]
